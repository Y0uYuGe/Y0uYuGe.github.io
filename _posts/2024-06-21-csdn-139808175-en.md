# Address-Related
Each qemu virtual machine is a process on the host; the process uses mmap to allocate 0x40000000 bytes of host virtual memory to serve as the VM's physical memory

GVA: guest virtual address (virtual address in the VM)
GPA: guest physical address (physical address in the VM)
HVA: host virtual address (virtual address on the host)
HPA: host physical address (physical address on the host)

The GVA is translated to a GPA through the guest's page tables; the GPA is actually an offset within the space mmap'd by the host process. Adding the base address gives the HVA (mmap allocates memory of the size requested by the VM, used as physical memory for that VM)




2. A device can request two types of space, memory mapped I/O (MMIO) and port mapped I/O (PMIO), and uses Base Address Registers (BAR) in the configuration space to mark the memory address information

```c
#include <stdio.h>
#include <string.h>
#include <stdint.h>
#include <stdlib.h>
#include <fcntl.h>
#include <assert.h>
#include <inttypes.h>

#define PAGE_SHIFT  12
#define PAGE_SIZE   (1 << PAGE_SHIFT)
#define PFN_PRESENT (1ull << 63)
#define PFN_PFN     ((1ull << 55) - 1)

int fd;

uint32_t page_offset(uint32_t addr)
{
    return addr & ((1 << PAGE_SHIFT) - 1);
    //虚拟地址addr在其所在页面内的偏移量
}

uint64_t gva_to_gfn(void *addr)
{
    uint64_t pme, gfn;
    size_t offset;
    offset = ((uintptr_t)addr >> 9) & ~7;//页表条目相对于文件开始的偏移量（GFN, Guest Physical Frame Number）
    lseek(fd, offset, SEEK_SET);//移动文件指针到计算出的偏移量处，准备读取PTE。
    read(fd, &pme, 8);  //从文件中读取8字节到pme变量即物理块起始地址
    if (!(pme & PFN_PRESENT))
        return -1;
    gfn = pme & PFN_PFN;
    return gfn;
}

uint64_t gva_to_gpa(void *addr)
{
    uint64_t gfn = gva_to_gfn(addr);
    assert(gfn != -1);
    return (gfn << PAGE_SHIFT) | page_offset((uint64_t)addr);
}

int main()
{
    uint8_t *ptr;
    uint64_t ptr_mem;
    // uint64_t ptr_malloc;
    fd = open("/proc/self/pagemap", O_RDONLY);
    if (fd < 0) {
        perror("open");
        exit(1);
    }

    ptr = malloc(256);
    strcpy(ptr, "Where am I?");
    printf("%s\n", ptr);
    ptr_mem = gva_to_gpa(ptr);
    
    printf("Your physical address is at 0x%"PRIx64"\n", ptr_mem);
    // ptr_malloc = gva_to_gpa(&malloc);
    // printf("malloc physical address 0x%"PRIx64" \n",ptr_malloc);

    getchar();
    return 0;
}
```

# Interaction-Related

 Each PCI device has a bus number, a device number, and a function identifier, and resides in a PCI domain. A PCI domain can host at most 256 buses, each bus can have at most 32 devices, and each device can have at most 8 functions 

   lspci can show the device's domain, bus number, device number, and function number; different versions of lspci display different content

   For data of the form 0000:00:03.0, 0000 is the domain, 00 is the bus number, 03 is the device number, and 0 is the function number
![在这里插入图片描述](/images/csdn/2024/csdn-139808175/image-004-d06ccfac58.png)
The pmio ports and mmio base address in resource are fixed
mmio: maps the configured registers to ports; access the ports to access the registers
pmio: maps the configured registers to memory; access the memory to access the registers


- hexdump /sys/devices/pci0000:00/0000:00:03.0/config to view configuration information such as deviceid and vendorid
- Accessing memory mapped I/O memory: open it to get an fd, then use mmap to map the memory address, and operate on the memory directly

```c
int mmio_fd = open("/sys/devices/pci0000:00/0000:00:04.0/resource0", O_RDWR | O_SYNC);
mmio_mem = mmap(0, 0x1000, PROT_READ | PROT_WRITE, MAP_SHARED, mmio_fd, 0);
```

- Accessing port mapped I/O memory: after requesting port access permission, access with in and out

```c
iopl(3);// iopl(3) 表示打开对全部IO端口的访问权限
inb(port); //从端口读一个字节
outb(val,port); //往端口写一个字节val
```

A device registers its own MMIO and PMIO read/write functions; when an access to its memory or ports is detected, the registered read/write functions are called, implementing read/write access to the device memory



When the memory where the device resides is accessed, the registered mmio read function d3dev_mmio_read is called; the read location is base address + 0x128, 4 bytes. It is recognized by d3dev_mmio_read as rsi = 0x128 (the offset) and rdx = 4

```c
#include <assert.h>
#include <fcntl.h>
#include <inttypes.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/mman.h>
#include <sys/types.h>
#include <unistd.h>
#include<sys/io.h>


unsigned char* mmio_mem;

void die(const char* msg)
{
    perror(msg);
    exit(-1);
}


void mmio_write(uint32_t addr, uint32_t value)
{
    *((uint32_t*)(mmio_mem + addr)) = value;
}

uint32_t mmio_read(uint32_t addr)
{
    return *((uint32_t*)(mmio_mem + addr));
}


int main(int argc, char *argv[])
{

    // Open and map I/O memory for the strng device
    int mmio_fd = open("/sys/devices/pci0000:00/0000:00:03.0/resource0", O_RDWR | O_SYNC);//O_SYNC标志表示同步I/O操作，确保对文件的写操作在返回前完成
    if (mmio_fd == -1)
        die("mmio_fd open failed");

    mmio_mem = mmap(0, 0x1000, PROT_READ | PROT_WRITE, MAP_SHARED, mmio_fd, 0);
    //PROT_READ | PROT_WRITE：指定映射区域的访问权限，这里允许读写。
    //MAP_SHARED：映射类型，共享映射意味着对映射区域的修改会反映到设备上，其他映射同一资源的进程也能看到修改。
    //mmio_fd：之前通过open获得的文件描述符。
    //0：文件偏移量，从资源的起始位置开始映射。
    if (mmio_mem == MAP_FAILED)
        die("mmap mmio_mem failed");

    printf("mmio_mem @ %p\n", mmio_mem);

    mmio_read(0x128);
    mmio_write(0x128, 1337);
    mmio_read(0x128);

}
```

```c
#include <assert.h>
#include <fcntl.h>
#include <inttypes.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/mman.h>
#include <sys/types.h>
#include <unistd.h>
#include<sys/io.h>

void die(const char* msg)
{
    perror(msg);
    exit(-1);
}


uint32_t pmio_base=0xc040;

uint32_t pmio_write(uint32_t addr, uint32_t value)
{
    outl(value,addr);
}

uint32_t pmio_read(uint32_t addr)
{
    return (uint32_t)inl(addr);
}

int main(int argc, char *argv[])
{


    if (iopl(3) !=0 )
        die("I/O permission is not enough");
    pmio_read(pmio_base+4);
    pmio_write(pmio_base+4,1);
    pmio_read(pmio_base+4);

}
```

- When running the program inside qemu, the operations involved in device read/write are executed by the qemu process, not by the virtual machine — they operate on the virtual addresses of the qemu process, not on the part mmap'd to serve as the guest's memory (just like normal user space)
- The addr in the registered read/write functions is an offset; the mmio range is 0x800, and pmio is 0x20

# Configuration-Related
A PCI device has a configuration space to store device information; the very first data in the header is the Device id and Vendor id

 When initializing a device, four important structures are initialized: TypeInfo -> TypeImpl -> ObjectClass -> Object. Each Object corresponds to a specific device, and its constructor is called when qemu starts and loads the device with the -device parameter

The first parameter of the device read/write operation functions is a pointer to this Object class



# Debugging

Package the exp into the filesystem

```bash
gcc -static exp.c -o   exp
find . | cpio -o --format=newc > ../rootfs.img
```
Start qemu

```bash
./launch.sh
```

```bash
touch .gdbinit
往里面写入断点
b*d3dev_pmio_write
b*d3dev_pmio_read
b*d3dev_mmio_write
b*d3dev_mmio_read
去除符号表就用
b *$rebase(addr)
```

```bash
ps -ef | grep qemu
一般是最下面的那个进程

```

```bash
sudo gdb attach -p 进程号
```

In qemu:

```bash
./exp
```
At this point gdb breaks at the breakpoints, and you can start debugging

> After a successful attach, you can set breakpoints and debug. PIE protection is enabled and there is no symbol table; you can use pwndbg's $rebase() to determine the real address in the process based on the address in IDA
