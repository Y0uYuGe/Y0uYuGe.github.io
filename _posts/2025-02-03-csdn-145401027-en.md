# Topics + Messages + Publishers

```cpp
#include<ros/ros.h>          // ROS的核心功能头文件
#include<std_msgs/String.h>  // 字符串消息类型的头文件
int main(int argc, char  *argv[])
{   

    ros::init(argc,argv,"chao_node");// 初始化ROS节点，节点名为"chao_node"
    ros::NodeHandle nh;  // 创建节点句柄，用于管理节点资源
    ros::Publisher pub = nh.advertise<std_msgs::String>("chao",10);

    ros::Rate loop_rate(10);// 设置循环频率为10Hz
    while(ros::ok())//使用ture时候运行节点不能响应外界信号
    {
        printf("keep\n");
        std_msgs::String msg; // 创建字符串消息对象
        msg.data = "do this all day man!!!"; // 设置消息内容
        pub.publish(msg);// 发布消息
        loop_rate.sleep(); // 控制循环频率
    }
    printf("hello world");
    return 0;
}

```


```python
#! /usr/bin/env python3
#coding=utf-8

import rospy
from std_msgs.msg import String
if __name__=="__main__":
    rospy.init_node("chao_node")
    pub=rospy.Publisher("chao",String,queue_size=10)
    rate=rospy.Rate(10)
    
    while not rospy.is_shutdown():
        msg=String()
        msg.data="do this all day man!!!"
        pub.publish(msg)
        rospy.loginfo("man!!!")
        rate.sleep()

```

# Subscribers

```cpp
#include<ros/ros.h>
#include<std_msgs/String.h>
void chao_callback(std_msgs::String msg)
{
    ROS_INFO(msg.data.c_str());
    printf("\n");
}
void yao_callback(std_msgs::String msg)
{
    ROS_WARN(msg.data.c_str());
    printf("\n");
}

int main(int argc, char  *argv[])
{   
    setlocale(LC_ALL,"zh_CN.UTF-8");
    ros::init(argc,argv,"ma_node");
    ros::NodeHandle nh;
    ros::Subscriber sub1 = nh.subscribe("chao",10,chao_callback);
    ros::Subscriber sub2 = nh.subscribe("yao",10,yao_callback);
    //10代表等待处理的缓存消息队列长度
    while(ros::ok())//使用ture时候运行节点不能响应外界信号
    {
        ros::spinOnce();
    }
    printf("hello world");
    return 0;
}

```

```python
#! /usr/bin/env python3
#coding=utf-8

import rospy
from std_msgs.msg import String

def yao_callback(msg):
    rospy.logwarn(msg.data)

if __name__=="__main__":
    rospy.init_node("ma_node")
    sub2=rospy.Subscriber("chao",String,chao_callback,queue_size=10)
    sub2=rospy.Subscriber("yao",String,yao_callback,queue_size=10)
    rospy.spin()

```

# Respective Buffer Queues + Overall Flow
In ROS (Robot Operating System), nodes communicate with each other through topics. A publisher is responsible for sending messages to a topic, while a subscriber is responsible for receiving messages from that topic. 10 is the queue size, which controls the buffering behavior of messages between the publisher and the subscriber.

For the following publisher and subscriber
```cpp
ros::Publisher pub = nh.advertise<std_msgs::String>("chao",10);
ros::Subscriber sub1 = nh.subscribe("chao",10,chao_callback);
```

# Robot Motion Control

```cpp
#include <ros/ros.h>
#include <geometry_msgs/Twist.h>

int main(int argc, char* argv[])
{   
    ros::init(argc,argv,"vel_node");
    ros::NodeHandle n;
    ros::Publisher vel_pub=n.advertise<geometry_msgs::Twist>("/cmd_vel",10);
    geometry_msgs::Twist vel_msg;
    vel_msg.linear.x=0.1;
    vel_msg.linear.y=0;
    vel_msg.linear.z=0;
    vel_msg.angular.x=0;
    vel_msg.angular.y=0;
    vel_msg.angular.z=0;

    ros::Rate r(30);
    while(ros::ok())
    {
        vel_pub.publish(vel_msg);
        r.sleep();
    }

    return 0;
}
```

```python
#! /usr/bin/env python3
#coding=utf-8

import rospy
from geometry_msgs.msg import Twist
if __name__=="__main__":
    rospy.init_node("vel_node")
    vel_pub=rospy.Publisher("cmd_vel",Twist,queue_size=10)
    vel_msg=Twist()
    vel_msg.linear.x=0.1
    rate=rospy.Rate(30)
    while not rospy.is_shutdown():
        vel_pub.publish(vel_msg)
        rate.sleep()

```
