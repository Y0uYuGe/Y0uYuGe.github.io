d3note libc 2.37

源码

索引可以越界，show 也没有检查，直接输出。在这里找到 GOT 表，然后开始查找。 然后寻找一个能存放 free 的 GOT 地址的 PIE 地址，而且这个 PIE 地址必须按 8 对齐；也不能再低，因为更低的 GOT 地址没有写权限——没找到。尝试寻找一个 PIE 地址，其中存放另一个保存着 chunk_array 的 PIE 地址——没找到。再找一个 PIE 地址，其中存放 chunk_array 下方的 PIE 地址——找到了，而且可写。然后修改 chunk_array 中的 len 和 ptr，实现任意地址读写，再写 GOT 表

修改 write

https://blog.xmcve.com/2024/04/29/D3CTF-2024-Writeup/#title-4 原本 write 会将 flag 数组某个偏移处的字节值，写入 open("/proc/self/mem", O_RDWR) 返回的文件描述符。不难想到，可以将文件描述符改为标准输出，让内容回显

unsigned __int64 __fastcall write(unsigned int a1, const char *a2, size_t a3)
{
  unsigned __int64 result; // rax
  unsigned __int64 v4; // rax
  unsigned int v5; // r8d
  unsigned __int64 fd; // [rsp+0h] [rbp-20h]

  if ( _libc_single_threaded )
  {
    result = sys_write(a1, a2, a3);
    if ( result > 0xFFFFFFFFFFFFF000LL )
    {
      __writefsdword((unsigned int)&errno, -(int)result);
      return -1LL;
    }
  }
  else
  {
    sub_938F0();
    v4 = sys_write(a1, a2, a3);
    if ( v4 > 0xFFFFFFFFFFFFF000LL )
    {
      __writefsdword((unsigned int)&errno, -(int)v4);
      v4 = -1LL;
    }
    fd = v4;
    sub_93970(v5);
    return fd;
  }
  return result;
}

这里根据 _libc_single_threaded 进入不同分支；如果 __libc_single_threaded 为零，就通过栈上某个偏移处的内容传递参数 服务端处理客户端连接时，会将 socket 文件描述符重定向到 0、1、2，使标准输入、标准输出和标准错误都指向同一个连接 socket。这样，客户端向 socket 发送的数据，会成为服务端程序从 0 读取的输入；客户端接收的数据，则来自服务端向 2 输出的内容，进而输出到 socket

.text:000000000011B2C4 8B 7C 24 08     mov     edi, dword ptr [rsp+28h+fd] ; fd

  1. 8B：这是 MOV 指令的操作码

    • 8B 表示从内存向寄存器进行 32 位数据传送

  2. 7C：这是 ModR/M 字节

    • 7 = 111b：表示使用 [寄存器 + 偏移] 寻址模式
    • C = 1100b:
      • 11b：表示目标操作数是寄存器
      • 00b：表示目标寄存器是 EDI（000 对应 EAX，001 对应 ECX，……，111 对应 EDI）

  3. 24：这是 SIB（比例因子-索引-基址）字节

    • 2 = 00 10b:
      • 00b：比例因子为 1
      • 10b：索引寄存器是 ESP（100）
    • 4 = 100b：基址寄存器是 ESP/RSP（100）

  4. 08：这是一个 8 位位移值

    • 表示相对于 RSP 的偏移为 8 字节

说明：

  • MOV 指令用于数据传送
  • 目标操作数是 EDI 寄存器
  • 源操作数是由 [RSP + 8] 指定的内存位置
  • 使用 SIB 字节是因为基址寄存器为 ESP/RSP

注意：

  1. 在 64 位模式下，即使操作数为 32 位，地址计算仍使用 64 位寄存器（这里是 RSP）
  2. 原始反汇编中的 [rsp+28h+fd] 在编译后被优化为 [rsp+8]，可能是因为编译器重新安排了栈帧布局

所以这里直接修改 08 这个字节，使 [rsp+8] 为零

.text:000000000011B284 80 3D B5 B2 0E 00 00  cmp     cs:__libc_single_threaded, 0

  1. 80：这是操作码，表示 cmp 指令。cmp 是比较指令，用于比较两个操作数。

  2. 3D：这是 cmp 指令的 ModR/M 字节，表示将立即数与某个内存地址处的值进行比较。

  3. **B5 B2 0E 00 **：这四个字节表示内存地址的偏移量。在这条指令中，内存地址是相对于当前指令位置的偏移。这是一个 32 位小端地址（低字节在前，高字节在后），对应偏移 0x000EB2B5。

  4. 00：这是用于比较的立即数，表示与 0 比较。

__libc_single_threaded 的值似乎默认为 1，这里将立即数改为 1

	int page_num;
	zend_mm_bin *bin;
	zend_mm_free_slot *p, *end;

	bin = (zend_mm_bin*)zend_mm_alloc_pages(heap, bin_pages[bin_num] ZEND_FILE_LINE_RELAY_CC ZEND_FILE_LINE_ORIG_RELAY_CC);

	if (UNEXPECTED(bin == NULL)) {
		/* insufficient memory */
		return NULL;
	}

	chunk = (zend_mm_chunk*)ZEND_MM_ALIGNED_BASE(bin, ZEND_MM_CHUNK_SIZE);
	page_num = ZEND_MM_ALIGNED_OFFSET(bin, ZEND_MM_CHUNK_SIZE) / ZEND_MM_PAGE_SIZE;
	chunk->map[page_num] = ZEND_MM_SRUN(bin_num);

		do {
			chunk->map[page_num+i] = ZEND_MM_NRUN(bin_num, i);
			i++;
		} while (i < bin_pages[bin_num]);
	}

	/* create a linked list of elements from 1 to last */
	end = (zend_mm_free_slot*)((char*)bin + (bin_data_size[bin_num] * (bin_elements[bin_num] - 1)));
	heap->free_slot[bin_num] = p = (zend_mm_free_slot*)((char*)bin + bin_data_size[bin_num]);

	p->next_free_slot = NULL;

	/* return first element */
	return bin;
}
